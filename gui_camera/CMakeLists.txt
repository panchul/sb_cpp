cmake_minimum_required(VERSION 3.16)
project(gui_camera_control LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(gui_camera_control camera_control.cpp)

find_package(SDL2 QUIET CONFIG)
if(TARGET SDL2::SDL2)
    target_link_libraries(gui_camera_control PRIVATE SDL2::SDL2)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED sdl2)
    target_include_directories(gui_camera_control PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_directories(gui_camera_control PRIVATE ${SDL2_LIBRARY_DIRS})
    target_compile_options(gui_camera_control PRIVATE ${SDL2_CFLAGS_OTHER})
    target_link_libraries(gui_camera_control PRIVATE ${SDL2_LIBRARIES})
endif()

find_package(SDL2_ttf QUIET CONFIG)
if(TARGET SDL2_ttf::SDL2_ttf)
    target_link_libraries(gui_camera_control PRIVATE SDL2_ttf::SDL2_ttf)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2_TTF REQUIRED SDL2_ttf)
    target_include_directories(gui_camera_control PRIVATE ${SDL2_TTF_INCLUDE_DIRS})
    target_link_directories(gui_camera_control PRIVATE ${SDL2_TTF_LIBRARY_DIRS})
    target_compile_options(gui_camera_control PRIVATE ${SDL2_TTF_CFLAGS_OTHER})
    target_link_libraries(gui_camera_control PRIVATE ${SDL2_TTF_LIBRARIES})
endif()

find_package(OpenCV QUIET COMPONENTS core imgproc videoio)
if(OpenCV_FOUND)
    target_include_directories(gui_camera_control PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(gui_camera_control PRIVATE ${OpenCV_LIBS})
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(OPENCV4 REQUIRED opencv4)
    target_include_directories(gui_camera_control PRIVATE ${OPENCV4_INCLUDE_DIRS})
    target_link_directories(gui_camera_control PRIVATE ${OPENCV4_LIBRARY_DIRS})
    target_compile_options(gui_camera_control PRIVATE ${OPENCV4_CFLAGS_OTHER})
    target_link_libraries(gui_camera_control PRIVATE ${OPENCV4_LIBRARIES})
endif()
